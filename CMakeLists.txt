cmake_minimum_required(VERSION 3.22)
project(obs-upscaler VERSION 0.2.0 LANGUAGES C)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an out-of-source build: cmake -S . -B build")
endif()

option(BUILD_PLUGIN "Build the OBS video filter" ON)
option(ENABLE_STRICT_WARNINGS "Treat warnings in this project as errors" OFF)
option(BUILD_OBS_SMOKE_TEST "Build a live libobs graphics test (requires a display)" OFF)
include(CTest)
include(GNUInstallDirs)
# The release SDK is the supported minimum, not a maximum runtime version.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/cmake/obs-sdk.json" OBS_SDK_SPEC)
string(JSON OBS_MINIMUM_VERSION GET "${OBS_SDK_SPEC}" version)
string(REPLACE "." ";" OBS_VERSION_PARTS "${OBS_MINIMUM_VERSION}")
list(GET OBS_VERSION_PARTS 0 OBS_MIN_MAJOR)
list(GET OBS_VERSION_PARTS 1 OBS_MIN_MINOR)
list(GET OBS_VERSION_PARTS 2 OBS_MIN_PATCH)
math(EXPR OBS_MINIMUM_API "(${OBS_MIN_MAJOR} << 24) | (${OBS_MIN_MINOR} << 16) | ${OBS_MIN_PATCH}")


function(upscaler_compile_options target)
  target_compile_features(${target} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /Zc:preprocessor)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    if(ENABLE_STRICT_WARNINGS)
      target_compile_options(${target} PRIVATE /WX)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    if(ENABLE_STRICT_WARNINGS)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
  endif()
endfunction()

if(BUILD_PLUGIN)
  # A custom SDK can be supplied through CMAKE_PREFIX_PATH or libobs_DIR.
  # Build releases against the minimum SDK; newer compatible runtimes are allowed.
  find_package(libobs CONFIG QUIET)
  if(NOT TARGET OBS::libobs)
    if(WIN32)
      message(FATAL_ERROR "OBS SDK not found. Run scripts/build-windows.ps1, or set libobs_DIR to your SDK's cmake directory.")
    endif()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LIBOBS REQUIRED IMPORTED_TARGET libobs)
    add_library(OBS::libobs ALIAS PkgConfig::LIBOBS)
  endif()
  if(DEFINED libobs_VERSION AND libobs_VERSION VERSION_LESS OBS_MINIMUM_VERSION)
    message(WARNING "This build uses OBS ${libobs_VERSION}. This is a development compatibility build below the supported minimum OBS ${OBS_MINIMUM_VERSION}.")
  endif()
  if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Windows package requires a 64-bit plugin. Configure with -A x64.")
  endif()
  if(APPLE)
    message(WARNING "macOS packaging and runtime compatibility are not validated in this release.")
  endif()

  add_library(obs-upscaler MODULE
    src/plugin-main.c
    src/upscale-filter.c
    src/upscale-policy.c)
  target_link_libraries(obs-upscaler PRIVATE OBS::libobs)
  if(NOT WIN32)
    target_link_libraries(obs-upscaler PRIVATE m)
  endif()
  target_compile_definitions(obs-upscaler PRIVATE PLUGIN_VERSION="${PROJECT_VERSION}"
    PLUGIN_MIN_OBS_VERSION="${OBS_MINIMUM_VERSION}" PLUGIN_MIN_OBS_API=${OBS_MINIMUM_API})
  set_target_properties(obs-upscaler PROPERTIES PREFIX "")
  upscaler_compile_options(obs-upscaler)

  if(WIN32)
    # Extract the resulting ZIP into C:\ProgramData\obs-studio\plugins.
    set(plugin_binary_dir "obs-upscaler/bin/64bit")
    set(plugin_data_dir "obs-upscaler/data")
    set(plugin_document_dir "obs-upscaler")
  else()
    set(plugin_binary_dir "${CMAKE_INSTALL_LIBDIR}/obs-plugins")
    set(plugin_data_dir "${CMAKE_INSTALL_DATADIR}/obs/obs-plugins/obs-upscaler")
    set(plugin_document_dir "${CMAKE_INSTALL_DOCDIR}")
  endif()
  install(TARGETS obs-upscaler
    RUNTIME DESTINATION "${plugin_binary_dir}"
    LIBRARY DESTINATION "${plugin_binary_dir}")
  install(DIRECTORY data/ DESTINATION "${plugin_data_dir}")
  install(FILES README.md LICENSE DESTINATION "${plugin_document_dir}")

  set(CPACK_PACKAGE_NAME "obs-upscaler")
  set(CPACK_PACKAGE_VENDOR "James C")
  set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
  set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
  set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}/packages")
  if(WIN32)
    set(CPACK_GENERATOR ZIP)
    set(CPACK_PACKAGE_FILE_NAME "obs-upscaler-${PROJECT_VERSION}-windows-x64")
  else()
    set(CPACK_GENERATOR TGZ)
    set(CPACK_PACKAGE_FILE_NAME "obs-upscaler-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
  endif()
  include(CPack)
endif()

if(BUILD_TESTING)
  add_executable(test-policy tests/test-policy.c src/upscale-policy.c)
  target_include_directories(test-policy PRIVATE src)
  upscaler_compile_options(test-policy)
  if(NOT WIN32)
    target_link_libraries(test-policy PRIVATE m)
  endif()
  add_test(NAME upscale-policy COMMAND test-policy)

  if(BUILD_OBS_SMOKE_TEST)
    if(NOT BUILD_PLUGIN)
      message(FATAL_ERROR "BUILD_OBS_SMOKE_TEST requires BUILD_PLUGIN=ON")
    endif()
    add_executable(smoke-obs tests/smoke-obs.c)
    target_link_libraries(smoke-obs PRIVATE OBS::libobs)
    if(WIN32)
      target_link_libraries(smoke-obs PRIVATE dxgi dxguid)
    else()
      target_link_libraries(smoke-obs PRIVATE m)
    endif()
    upscaler_compile_options(smoke-obs)
    add_dependencies(smoke-obs obs-upscaler)
    add_test(NAME obs-smoke
      COMMAND smoke-obs "$<TARGET_FILE:obs-upscaler>" "${CMAKE_SOURCE_DIR}/data")
    set_tests_properties(obs-smoke PROPERTIES TIMEOUT 120)
  endif()
endif()
